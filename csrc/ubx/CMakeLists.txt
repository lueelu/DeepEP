cmake_minimum_required(VERSION 3.20)
project(deepep_ubx LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
find_package(pybind11 CONFIG REQUIRED)
set(DEEPEP_UBX_RANKS_PER_HOST 16 CACHE STRING "Participating UBX ranks per host")
if(NOT DEEPEP_UBX_RANKS_PER_HOST MATCHES "^(4|16)$")
    message(FATAL_ERROR "UBX requires 4 or 16 participating ranks per host")
endif()
if(NOT TARGET shmem)
    include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/Dependencies.cmake")
endif()
# All machine backends use the repository-wide SHMEM lock.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../dependencies.lock.json" _lock)
string(JSON _revision GET "${_lock}" shmem revision)
file(READ "${SHMEM_ROOT}/deepep-sdk.json" _sdk)
string(JSON _head GET "${_sdk}" revision)
if(NOT _head STREQUAL _revision)
    message(FATAL_ERROR "UBX requires SHMEM release/v1.8.0 at ${_revision}")
endif()
string(JSON _backend GET "${_sdk}" build rdma_backend)
string(JSON _pi_mode GET "${_sdk}" build hns_1825_pi_on_chip)
if(NOT _backend STREQUAL "HNS_1825" OR NOT _pi_mode MATCHES "^[01]$")
    message(FATAL_ERROR "Rebuild SHMEM with HNS_1825 and record the detected doorbell mode")
endif()
# Internal WQE helpers are part of this operator's dependency contract.
string(JSON _count LENGTH "${_lock}" shmem device_header_sha256_lf)
math(EXPR _last "${_count} - 1")
foreach(_index RANGE ${_last})
    string(JSON _header MEMBER "${_lock}" shmem device_header_sha256_lf ${_index})
    string(JSON _expected GET "${_lock}" shmem device_header_sha256_lf "${_header}")
    file(READ "${SHMEM_ROOT}/${_header}" _installed_text)
    string(REPLACE "\r\n" "\n" _installed_text "${_installed_text}")
    string(SHA256 _actual "${_installed_text}")
    if(NOT _actual STREQUAL _expected)
        message(FATAL_ERROR "Installed UBX SDK differs from its pinned source: ${_header}")
    endif()
endforeach()
find_path(_platform_include tiling/platform/platform_ascendc.h HINTS
    "${ASCEND_HOME_PATH}/include" "${ASCEND_HOME_PATH}/asc/include"
    "${ASCEND_HOME_PATH}/${CMAKE_SYSTEM_PROCESSOR}-linux/asc/include" NO_DEFAULT_PATH REQUIRED)
find_library(_tiling NAMES tiling_api PATHS ${_cann_library_paths} NO_DEFAULT_PATH REQUIRED)
find_library(_platform NAMES platform PATHS ${_cann_library_paths} NO_DEFAULT_PATH REQUIRED)
set(_device "${CMAKE_CURRENT_BINARY_DIR}/device")
add_custom_target(ubx_device_build
    COMMAND "${CMAKE_COMMAND}" -S "${CMAKE_CURRENT_SOURCE_DIR}/kernels" -B "${_device}"
        "-DCMAKE_CXX_COMPILER=${ASCEND_HOME_PATH}/tools/bisheng_compiler/bin/bisheng"
        "-DASCEND_HOME_PATH=${ASCEND_HOME_PATH}" "-DSHMEM_ROOT=${SHMEM_ROOT}"
        "-DDEEPEP_UBX_RANKS_PER_HOST=${DEEPEP_UBX_RANKS_PER_HOST}"
        "-DDEEPEP_UBX_PI_ON_CHIP=${_pi_mode}"
    COMMAND "${CMAKE_COMMAND}" --build "${_device}" --parallel 4
    BYPRODUCTS "${_device}/libdeepep_ubx_kernels.so" VERBATIM)
add_library(ubx_device SHARED IMPORTED)
set_target_properties(ubx_device PROPERTIES IMPORTED_LOCATION "${_device}/libdeepep_ubx_kernels.so")
add_dependencies(ubx_device ubx_device_build)
pybind11_add_module(_ubx_C NO_EXTRAS bindings.cpp session.cpp)
target_include_directories(_ubx_C PRIVATE "${ASCEND_HOME_PATH}/include" "${SHMEM_ROOT}/include" "${_platform_include}")
target_compile_definitions(_ubx_C PRIVATE DEEPEP_UBX_RANKS_PER_HOST=${DEEPEP_UBX_RANKS_PER_HOST})
target_compile_options(_ubx_C PRIVATE -O2 -Wall -Wextra)
target_link_libraries(_ubx_C PRIVATE ascendcl shmem ubx_device "${_tiling}" "${_platform}" ${CMAKE_DL_LIBS})
set_target_properties(_ubx_C PROPERTIES INSTALL_RPATH "$ORIGIN" INSTALL_RPATH_USE_LINK_PATH FALSE)
install(TARGETS _ubx_C LIBRARY DESTINATION .)
install(FILES "${_device}/libdeepep_ubx_kernels.so" DESTINATION .)
file(SHA256 "${SHMEM_LIBRARY}" _shmem_sha)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ubx-build.json"
    "{\"shmem_revision\":\"${_revision}\",\"shmem_sha256\":\"${_shmem_sha}\",\"ranks_per_host\":${DEEPEP_UBX_RANKS_PER_HOST}}\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/ubx-build.json" DESTINATION .)
